// ─────────────────────────────────────────────────────────────
//  firestore.rules
//  Paste this entire block into:
//  Firebase Console → Firestore Database → Rules → Edit rules
//
//  BEFORE PASTING: replace 2z9Z3K5ZwShvadUuqZmwMv0s1Od2 with your actual UID.
//  Find it at: Firebase Console → Authentication → Users →
//  click your user → copy the User UID value shown.
// ─────────────────────────────────────────────────────────────

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isAdmin() {
      return request.auth != null
          && request.auth.uid == "2z9Z3K5ZwShvadUuqZmwMv0s1Od2";
    }

    function userDoc(uid) {
      return get(/databases/$(database)/documents/users/$(uid)).data;
    }

    // users/{uid} fields that grant access or identity. Only the admin
    // (Manage Users console) and Cloud Functions (admin SDK) may change
    // them: the esFormLab / esMetabolicLab / coaching paywalls and
    // verifyActivePass() trust features + formAnalyzerPassUntil, the
    // coaching dashboard keys off role, coachUid decides who can read the
    // athlete, and the Strava webhook resolves users by stravaAthleteId.
    function protectedUserFields() {
      return ['role', 'features', 'formAnalyzerPassUntil', 'coachUid',
              'stravaAthleteId', 'stravaConnected'];
    }

    // A user doc created by anyone but the admin starts with no
    // entitlements, the default role and no Strava link.
    function isPlainNewUser(data) {
      return !data.keys().hasAny(['features', 'formAnalyzerPassUntil'])
          && data.get('role', 'athlete') == 'athlete'
          && data.get('stravaAthleteId', null) == null
          && data.get('stravaConnected', false) == false;
    }

    // ── Computed ranges (male + female) ──────────────────────
    // Anyone can READ — the public esFormLab analyzer fetches
    // computed_ranges/male and computed_ranges/female on load.
    // Only the admin account can WRITE.
    match /computed_ranges/{sex} {
      allow read:  if true;
      allow write: if request.auth != null
                   && request.auth.uid == "2z9Z3K5ZwShvadUuqZmwMv0s1Od2"
                   && sex in ["male", "female", "combined"];
    }

    // ── Reference sessions ───────────────────────────────────
    // Only the admin can read or write raw session data.
    match /reference_sessions/{sessionId} {
      allow read, write: if request.auth != null
                         && request.auth.uid == "2z9Z3K5ZwShvadUuqZmwMv0s1Od2";
    }

    // ── Phase labels ─────────────────────────────────────────
    // Admin-supplied ground-truth labels for gait-phase frame detection.
    // Only the admin can read or write; used to tune detection scoring.
    match /phase_labels/{labelId} {
      allow read, write: if request.auth != null
                         && request.auth.uid == "2z9Z3K5ZwShvadUuqZmwMv0s1Od2";
    }

    // ── User profiles ────────────────────────────────────────
    // Users can read/write their own profile, except the protected
    // fields above. Coach can read/edit profiles of athletes assigned to
    // them (same restriction). Admin can do anything.
    match /users/{userId} {
      allow read: if request.auth != null
                  && (request.auth.uid == userId
                      || resource.data.coachUid == request.auth.uid
                      || request.auth.uid == "2z9Z3K5ZwShvadUuqZmwMv0s1Od2");
      // Your own doc (first sign-in, or any set-merge on a missing doc),
      // or -- as a coach -- an athlete account you're adding to your own
      // roster (coaching dashboard "Add athlete").
      allow create: if isAdmin()
                    || (request.auth != null
                        && isPlainNewUser(request.resource.data)
                        && ((request.auth.uid == userId
                             && request.resource.data.get('coachUid', null) == null)
                            || (request.resource.data.get('coachUid', null) == request.auth.uid
                                && userDoc(request.auth.uid).get('role', '') == 'coach')));
      allow update: if isAdmin()
                    || (request.auth != null
                        && (request.auth.uid == userId
                            || request.auth.uid == resource.data.get('coachUid', null))
                        && !request.resource.data.diff(resource.data).affectedKeys()
                              .hasAny(protectedUserFields()));

      // ── Saved analyses subcollection ─────────────────────────
      match /analyses/{analysisId} {
        allow read, write: if request.auth != null
                           && request.auth.uid == userId;
        // Admin can read, and write on the athlete's behalf ("act as")
        allow read, write: if request.auth != null
                           && request.auth.uid == "2z9Z3K5ZwShvadUuqZmwMv0s1Od2";
      }

      // ── Garmin activities subcollection ─────────────────────
      // Backfilled from a Garmin Connect CSV export by the Run
      // Dynamics tool. The athlete, their assigned coach, and the
      // admin can all read and write (coach may backfill on behalf
      // of the athlete from the coaching dashboard).
      match /garminActivities/{activityId} {
        allow read, write: if request.auth != null
                           && (request.auth.uid == userId
                               || request.auth.uid == "2z9Z3K5ZwShvadUuqZmwMv0s1Od2"
                               || get(/databases/$(database)/documents/users/$(userId)).data.coachUid == request.auth.uid);
      }

      // ── Private data (Strava tokens etc) ────────────────────
      // Only Cloud Functions (admin SDK) can access — never the client.
      match /private/{docId} {
        allow read, write: if false;
      }
    }

    // ── Training plans ───────────────────────────────────────
    match /trainingPlans/{planId} {
      // Coach who owns the plan can read/write.
      // Athlete the plan belongs to can read.
      // Admin can read any plan (view-only across all athletes).
      allow read: if request.auth != null
                  && (resource.data.coachUid == request.auth.uid
                      || resource.data.athleteUid == request.auth.uid
                      || request.auth.uid == "2z9Z3K5ZwShvadUuqZmwMv0s1Od2");
      // Only a coach for an athlete already assigned to them (or the
      // admin). The Strava sync and webhook attach activities to the
      // athlete's first plan (athleteUid == uid, limit 1), so a plan that
      // names someone else as athlete would capture their Strava data.
      allow create: if isAdmin()
                    || (request.auth != null
                        && request.resource.data.coachUid == request.auth.uid
                        && userDoc(request.resource.data.athleteUid).get('coachUid', null) == request.auth.uid);
      allow update: if request.auth != null
                    && resource.data.coachUid == request.auth.uid
                    && !request.resource.data.diff(resource.data).affectedKeys()
                          .hasAny(['athleteUid', 'coachUid']);

      // ── Workouts subcollection ─────────────────────────────
      match /workouts/{workoutId} {
        allow read: if request.auth != null
                    && (get(/databases/$(database)/documents/trainingPlans/$(planId)).data.coachUid == request.auth.uid
                        || get(/databases/$(database)/documents/trainingPlans/$(planId)).data.athleteUid == request.auth.uid
                        || request.auth.uid == "2z9Z3K5ZwShvadUuqZmwMv0s1Od2");
        // Coach can fully write; athlete can write (to add their notes)
        allow write: if request.auth != null
                     && (get(/databases/$(database)/documents/trainingPlans/$(planId)).data.coachUid == request.auth.uid
                         || get(/databases/$(database)/documents/trainingPlans/$(planId)).data.athleteUid == request.auth.uid);
      }

      // ── Strava activities subcollection ────────────────────
      match /stravaActivities/{activityId} {
        allow read: if request.auth != null
                    && (get(/databases/$(database)/documents/trainingPlans/$(planId)).data.coachUid == request.auth.uid
                        || get(/databases/$(database)/documents/trainingPlans/$(planId)).data.athleteUid == request.auth.uid
                        || request.auth.uid == "2z9Z3K5ZwShvadUuqZmwMv0s1Od2");
        // Athlete can write their own Strava data (or a Cloud Function)
        allow write: if request.auth != null
                     && get(/databases/$(database)/documents/trainingPlans/$(planId)).data.athleteUid == request.auth.uid;
      }

      // ── Lifting weeks subcollection ───────────────────────
      match /liftingWeeks/{weekId} {
        allow read: if request.auth != null
                    && (get(/databases/$(database)/documents/trainingPlans/$(planId)).data.coachUid == request.auth.uid
                        || get(/databases/$(database)/documents/trainingPlans/$(planId)).data.athleteUid == request.auth.uid
                        || request.auth.uid == "2z9Z3K5ZwShvadUuqZmwMv0s1Od2");
        // Both coach and athlete can write lifting data
        allow write: if request.auth != null
                     && (get(/databases/$(database)/documents/trainingPlans/$(planId)).data.coachUid == request.auth.uid
                         || get(/databases/$(database)/documents/trainingPlans/$(planId)).data.athleteUid == request.auth.uid);
      }
    }

    // ── Notifications ──────────────────────────────────────
    match /notifications/{notifId} {
      // Users can read their own notifications
      allow read: if request.auth != null
                  && resource.data.recipientUid == request.auth.uid;
      // ...and mark them read. Accept/decline also rewrites status +
      // message. Nothing can be re-addressed or re-typed.
      allow update: if request.auth != null
                    && resource.data.recipientUid == request.auth.uid
                    && request.resource.data.diff(resource.data).affectedKeys()
                         .hasOnly(['read', 'status', 'message']);
      // Only the notifications the site actually sends: feedback and
      // in-person requests to the admin (from yourself), a coaching
      // request for a request you filed, and coach -> their own athlete.
      // Admin can send anything; Cloud Functions use the admin SDK.
      // Client ids come from .add() (20 alphanumerics) -- reject any other
      // doc id so a crafted one can't reach the admin's notification list.
      allow create: if isAdmin()
                    || (request.auth != null
                        && notifId.matches('^[A-Za-z0-9]{20}$')
                        && ((request.resource.data.recipientUid == "2z9Z3K5ZwShvadUuqZmwMv0s1Od2"
                             && request.resource.data.type in ['bug_report', 'contact', 'in_person_request']
                             && request.resource.data.get('senderUid', null) == request.auth.uid)
                            || (request.resource.data.recipientUid == "2z9Z3K5ZwShvadUuqZmwMv0s1Od2"
                                && request.resource.data.type == 'coaching_request'
                                && get(/databases/$(database)/documents/coachingRequests/$(request.resource.data.requestId)).data.athleteUid == request.auth.uid)
                            || (request.resource.data.type in ['workout_updated', 'coaching_accepted', 'coaching_declined']
                                && userDoc(request.resource.data.recipientUid).get('coachUid', null) == request.auth.uid)));
    }

    // ── Coaching requests ────────────────────────────────────
    match /coachingRequests/{requestId} {
      // Athlete can read/create their own requests. The admin can read any
      // request (incl. a deleted one: get() then returns exists=false, so
      // the bell can tell a withdrawn/deleted request from a live one).
      allow read: if isAdmin()
                  || (request.auth != null
                      && (resource.data.athleteUid == request.auth.uid
                          || resource.data.coachUid == request.auth.uid));
      allow create: if isAdmin()
                    || (request.auth != null
                        && request.resource.data.athleteUid == request.auth.uid);
      // Coach can update (accept/decline)
      allow update: if request.auth != null
                    && resource.data.coachUid == request.auth.uid;
    }

    // ── Stripe: customer-scoped collections ─────────────────
    // Managed by the "Run Payments with Stripe" extension.
    // - customers/{uid}: customer record, created by the extension
    //   via admin SDK when a user signs in. Readable by that user.
    // - checkout_sessions: client writes a doc here to start a
    //   Stripe Checkout; extension writes back the hosted URL.
    // - subscriptions / payments: written by the extension via
    //   webhook; the client reads them to check pass status.
    match /customers/{uid} {
      allow read: if request.auth != null && request.auth.uid == uid;

      match /checkout_sessions/{sessionId} {
        allow read, write: if request.auth != null && request.auth.uid == uid;
      }

      match /subscriptions/{subId} {
        allow read: if request.auth != null && request.auth.uid == uid;
      }

      match /payments/{paymentId} {
        allow read: if request.auth != null && request.auth.uid == uid;
      }
    }

    // ── Stripe: public product catalog ──────────────────────
    // Mirror of Stripe products / prices / tax_rates, written
    // by the extension. Public read so the upgrade UI can show
    // pricing without requiring sign-in.
    match /products/{productId} {
      allow read: if true;

      match /prices/{priceId} {
        allow read: if true;
      }

      match /tax_rates/{taxRateId} {
        allow read: if true;
      }
    }

    // ── Deny everything else ─────────────────────────────────
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
